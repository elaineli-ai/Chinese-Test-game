            osc.frequency.exponentialRampToValueAtTime(100, now + 0.2);
            gain.gain.setValueAtTime(0.3, now);
            gain.gain.exponentialRampToValueAtTime(0.01, now + 0.2);
            osc.start(now); osc.stop(now + 0.2);
        } else if (type === 'victory') { // 勝利凱旋音
            osc.frequency.setValueAtTime(523.25, now); // C5
            osc.frequency.setValueAtTime(523.25, now + 0.1);
            osc.frequency.setValueAtTime(523.25, now + 0.2);
            osc.frequency.setValueAtTime(659.25, now + 0.3); // E5
            osc.frequency.setValueAtTime(783.99, now + 0.4); // G5
            gain.gain.setValueAtTime(0.4, now);
            gain.gain.exponentialRampToValueAtTime(0.01, now + 0.7);
            osc.start(now); osc.stop(now + 0.7);
        }
    }

    // 初始化玩家狀態
    function initPlayerState(id) {
        return {
            id: id,
            hp: 100,
            maxHp: 100,
            score: 0,
            combo: 0,
            rightCount: 0,
            wrongCount: 0,
            killCount: 0,
            currentMonster: { hp: 50, maxHp: 50, name: "小怪獸", sprite: "👾" },
            qIdxList: [], // 隨洗題目索引
            currentQIndex: 0,
            userAnswers: []
        };
    }

    // 開始遊戲
    function startGame() {
        document.getElementById('start-screen').classList.add('hidden');
        p1State = initPlayerState(1);
        p2State = initPlayerState(2);
        
        // 洗牌題目
        p1State.qIdxList = [...Array(questionBank.length).keys()].sort(() => Math.random() - 0.5);
        p2State.qIdxList = [...Array(questionBank.length).keys()].sort(() => Math.random() - 0.5);

        // 生成按鍵區
        buildInputArea(1);
        buildInputArea(2);

        // 載入第一題與怪獸
        spawnMonster(p1State);
        spawnMonster(p2State);
        loadQuestion(p1State);
        loadQuestion(p2State);

        updateVisuals(p1State);
        updateVisuals(p2State);
    }

    // 生成輸入介面 (12個標點符號 + 2個功能鈕)
    function buildInputArea(pId) {
        const container = document.getElementById(`p${pId}-input-area`);
        container.innerHTML = "";
        
        // 標點按鈕
        punctButtons.forEach(p => {
            const btn = document.createElement('button');
            btn.className = "btn-punct";
            btn.innerText = p;
            btn.onclick = () => handlePunctClick(pId, p);
            container.appendChild(btn);
        });

        // 取消清除鈕
        const clearBtn = document.createElement('button');
        clearBtn.className = "btn-punct btn-control btn-clear";
        clearBtn.innerText = "清除";
        clearBtn.onclick = () => handleClear(pId);
        container.appendChild(clearBtn);

        // 確定送出鈕
        const submitBtn = document.createElement('button');
        submitBtn.className = "btn-punct btn-control btn-submit";
        submitBtn.innerText = "確定";
        submitBtn.onclick = () => handleSubmit(pId);
        container.appendChild(submitBtn);
    }

    // 召喚新怪獸
    function spawnMonster(pState) {
        const mIdx = Math.min(pState.killCount, monsterSprites.length - 1);
        pState.currentMonster.sprite = monsterSprites[mIdx];
        pState.currentMonster.name = monsterNames[mIdx];
        pState.currentMonster.maxHp = 50 + pState.killCount * 25; // 越後面血量越高
        pState.currentMonster.hp = pState.currentMonster.maxHp;
        
        document.getElementById(`m${pState.id}-sprite`).innerText = pState.currentMonster.sprite;
        document.getElementById(`m${pState.id}-name`).innerText = pState.currentMonster.name;
    }

    // 載入題目
    function loadQuestion(pState) {
        const qIdx = pState.qIdxList[pState.currentQIndex];
        pState.currentQuestion = questionBank[qIdx];
        pState.userAnswers = new Array(pState.currentQuestion.a.length).fill("");
        
        renderQuestionBox(pState);
    }

    // 渲染題目文字與填空處
    function renderQuestionBox(pState) {
        const qBox = document.getElementById(`p${pState.id}-q-box`);
        const textParts = pState.currentQuestion.q.split("___");
        qBox.innerHTML = "";

        // 尋找當前該填寫哪一個格子的索引
        let activeSlotIdx = pState.userAnswers.findIndex(ans => ans === "");
        if (activeSlotIdx === -1) activeSlotIdx = pState.userAnswers.length - 1;

        for (let i = 0; i < textParts.length; i++) {
            qBox.appendChild(document.createTextNode(textParts[i]));
            if (i < pState.currentQuestion.a.length) {
                const slot = document.createElement('span');
                slot.className = `q-slot ${i === activeSlotIdx ? 'active' : ''}`;
                slot.innerText = pState.userAnswers[i] || "";
                qBox.appendChild(slot);
            }
        }
    }

    // 處理標點點擊事件
    function handlePunctClick(pId, punct) {
        const pState = (pId === 1) ? p1State : p2State;
        // 尋找第一個還沒填空的格子
        const emptyIdx = pState.userAnswers.findIndex(ans => ans === "");
        if (emptyIdx !== -1) {
            pState.userAnswers[emptyIdx] = punct;
            renderQuestionBox(pState);
        }
    }

    // 處理清除按鈕
    function handleClear(pId) {
        const pState = (pId === 1) ? p1State : p2State;
        pState.userAnswers = new Array(pState.currentQuestion.a.length).fill("");
        renderQuestionBox(pState);
    }

    // 處理確定送出答案
    function handleSubmit(pId) {
        const pState = (pId === 1) ? p1State : p2State;
        
        // 檢查是否每個格子都有填寫
        if (pState.userAnswers.includes("")) {
            alert("請把所有格子的標點符號填完再送出喔！");
            return;
        }

        // 判斷對錯
        let isCorrect = true;
        for (let i = 0; i < pState.currentQuestion.a.length; i++) {
            if (pState.userAnswers[i] !== pState.currentQuestion.a[i]) {
                isCorrect = false;
                break;
            }
        }

        if (isCorrect) {
            // 答對邏輯
            playSound('correct');
            pState.rightCount++;
            pState.combo++;
            
            // 計算攻擊力 (基礎 25 點，依據連擊疊加)
            let dmg = 25;
            let comboBonusTxt = "";
            if (pState.combo >= 3) {
                dmg += 15;
                // 連擊補血機制
                pState.hp = Math.min(pState.maxHp, pState.hp + 20);
                showPopEffect(pState.id, 'player', '+20 HP', 'heal-pop');
                comboBonusTxt = " (💥Combo加成!)";
            }
            pState.score += (10 + pState.combo * 2);

            // 扣減怪獸血量
            pState.currentMonster.hp = Math.max(0, pState.currentMonster.hp - dmg);
            showPopEffect(pState.id, 'monster', `-${dmg}${comboBonusTxt}`);
            
            // 觸發攻擊與受傷動畫
            triggerAnimate(pState.id, 'attack');

            // 檢查怪獸是否死亡
            if (pState.currentMonster.hp === 0) {
                pState.killCount++;
                setTimeout(() => {
                    spawnMonster(pState);
                    updateVisuals(pState);
                }, 500);
            }
        } else {
            // 答錯邏輯
            playSound('wrong');
            pState.wrongCount++;
            pState.combo = 0; // 中斷連擊
            
            // 玩家扣血
            const dmgToPlayer = 20;
            pState.hp = Math.max(0, pState.hp - dmgToPlayer);
            showPopEffect(pState.id, 'player', `-${dmgToPlayer}`);
            triggerAnimate(pState.id, 'hurt');

            // 檢查玩家是否戰敗
            if (pState.hp === 0) {
                endGame();
                return;
            }
        }

        // 前進到下一題
        pState.currentQIndex++;
        // 題庫若播完了就重新洗牌重來
        if (pState.currentQIndex >= pState.qIdxList.length) {
            pState.qIdxList = [...Array(questionBank.length).keys()].sort(() => Math.random() - 0.5);
            pState.currentQIndex = 0;
        }

        loadQuestion(pState);
        updateVisuals(pState);
    }

    // 觸發物理動畫效果
    function triggerAnimate(pId, type) {
        const pSprite = document.getElementById(`p${pId}-sprite`);
        const mSprite = document.getElementById(`m${pId}-sprite`);
        
        if (type === 'attack') {
            playSound('attack');
            pSprite.classList.add(pId === 1 ? 'attack-animation' : 'p2-attack');
            setTimeout(() => pSprite.classList.remove('attack-animation', 'p2-attack'), 150);
            
            setTimeout(() => {
                mSprite.classList.add('hurt-animation');
                setTimeout(() => mSprite.classList.remove('hurt-animation'), 300);
            }, 100);
        } else if (type === 'hurt') {
            mSprite.classList.add(pId === 1 ? 'p2-attack' : 'attack-animation'); // 換怪獸撞玩家
            setTimeout(() => mSprite.classList.remove('attack-animation', 'p2-attack'), 150);

            setTimeout(() => {
                pSprite.classList.add('hurt-animation');
                setTimeout(() => pSprite.classList.remove('hurt-animation'), 300);
            }, 100);
        }
    }

    // 顯示漂浮文字傷害數字
    function showPopEffect(pId, target, text, customClass = '') {
        const arena = document.querySelectorAll('.rpg-arena')[pId - 1];
        const pop = document.createElement('div');
        pop.className = `damage-pop ${customClass}`;
        pop.innerText = text;
        
        if (target === 'player') {
            pop.style.left = '25%'; pop.style.top = '30%';
        } else {
            pop.style.left = '75%'; pop.style.top = '30%';
        }
        
        arena.appendChild(pop);
        setTimeout(() => pop.remove(), 800);
    }

    // 更新畫面狀態顯示 (血量條、分數等)
    function updateVisuals(pState) {
        const id = pState.id;
        // 分數與連擊
        document.getElementById(`p${id}-score`).innerText = pState.score;
        const comboTag = document.getElementById(`p${id}-combo-tag`);
        if (pState.combo >= 3) {
            comboTag.innerText = `${pState.combo} 連擊 🔥`;
            comboTag.style.display = "inline-block";
        } else {
            comboTag.style.display = "none";
        }

        // 玩家血條
        const pHpPercent = (pState.hp / pState.maxHp) * 100;
        document.getElementById(`p${id}-hp-bar`).style.width = `${pHpPercent}%`;
        document.getElementById(`p${id}-hp-txt`).innerText = `${pState.hp}/${pState.maxHp}`;

        // 怪獸血條
        const mHpPercent = (pState.currentMonster.hp / pState.currentMonster.maxHp) * 100;
        document.getElementById(`m${id}-hp-bar`).style.width = `${mHpPercent}%`;
        document.getElementById(`m${id}-hp-txt`).innerText = `${pState.currentMonster.hp}/${pState.currentMonster.maxHp}`;
    }

    // 遊戲結束結算
    function endGame() {
        playSound('victory');
        document.getElementById('end-screen').classList.remove('hidden');

        // 寫入結算數據
        document.getElementById('r-p1-score').innerText = p1State.score;
        document.getElementById('r-p1-right').innerText = p1State.rightCount;
        document.getElementById('r-p1-wrong').innerText = p1State.wrongCount;
        document.getElementById('r-p1-kills').innerText = p1State.killCount;

        document.getElementById('r-p2-score').innerText = p2State.score;
        document.getElementById('r-p2-right').innerText = p2State.rightCount;
        document.getElementById('r-p2-wrong').innerText = p2State.wrongCount;
        document.getElementById('r-p2-kills').innerText = p2State.killCount;

        // 判斷勝負
        document.getElementById('p1-win-tag').classList.add('hidden');
        document.getElementById('p2-win-tag').classList.add('hidden');
        if (p1State.score > p2State.score) {
            document.getElementById('p1-win-tag').classList.remove('hidden');
        } else if (p2State.score > p1State.score) {
            document.getElementById('p2-win-tag').classList.remove('hidden');
        }
    }

    function resetGame() {
        document.getElementById('end-screen').classList.add('hidden');
        startGame();
    }

    // 靜音控制
    function toggleMute() {
        isMuted = !isMuted;
        document.getElementById('mute-btn').innerText = isMuted ? "🔇 聲音：關" : "🔊 聲音：開";
    }

    // 電子白板全螢幕功能
    function toggleFullscreen() {
        const elem = document.getElementById('game-container');
        if (!document.fullscreenElement) {
            if (elem.requestFullscreen) elem.requestFullscreen();
            else if (elem.webkitRequestFullscreen) elem.webkitRequestFullscreen();
            else if (elem.msRequestFullscreen) elem.msRequestFullscreen();
        } else {
            if (document.exitFullscreen) document.exitFullscreen();
        }
    }
