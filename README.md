    const MONSTERS = [
      {name:"小龍",emoji:"🐲",hp:60},

    const MAX_HP = 100;

    const screens = ["start", "info", "game", "end"];

    let players = [];

    let liveOscillators = new Set();
    let finishTimer = null;

    function showScreen(name) {
      screens.forEach(s => document.getElementById(s + "-screen").hidden = s !== name);
    }
    function shuffledQuestions() {
      const copy = [...QUESTIONS];
      for (let i = copy.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [copy[i], copy[j]] = [copy[j], copy[i]];
      }
      return copy;
    }
    function stopSounds() {
      liveOscillators.forEach(o => { try { o.stop(); } catch (_) {} });
      liveOscillators.clear();
    }
    function tone(frequency, start, duration, type = "sine", volume = .12) {
      if (muted) return;
      try {
        audioContext ||= new (window.AudioContext || window.webkitAudioContext)();
        if (audioContext.state === "suspended") audioContext.resume();
        const oscillator = audioContext.createOscillator();
        const gain = audioContext.createGain();
        oscillator.type = type;
        oscillator.frequency.setValueAtTime(frequency, audioContext.currentTime + start);
        gain.gain.setValueAtTime(.001, audioContext.currentTime + start);
        gain.gain.exponentialRampToValueAtTime(volume, audioContext.currentTime + start + .015);
        gain.gain.exponentialRampToValueAtTime(.001, audioContext.currentTime + start + duration);
        oscillator.connect(gain).connect(audioContext.destination);
        oscillator.onended = () => liveOscillators.delete(oscillator);
        liveOscillators.add(oscillator);
        oscillator.start(audioContext.currentTime + start);
        oscillator.stop(audioContext.currentTime + start + duration + .01);
      } catch (_) { /* The game remains playable if audio is unavailable. */ }
    }
    function sound(kind) {
      if (kind === "correct") { tone(520, 0, .11); tone(740, .13, .17); }
      if (kind === "wrong") { tone(300, 0, .18, "triangle"); tone(210, .18, .22, "triangle"); }
      if (kind === "attack") { tone(180, 0, .12, "sawtooth", .075); tone(110, .1, .18, "triangle"); }
      if (kind === "victory") [440, 554, 659, 880].forEach((n, i) => tone(n, i * .13, .21));
    }
    function delay(ms) { return new Promise(resolve => setTimeout(resolve, ms)); }
    function animate(el, name, ms) {
      el.classList.remove(name);
      void el.offsetWidth;
      el.classList.add(name);
      setTimeout(() => el.classList.remove(name), ms);
    }
    function monsterFor(index) {
      const base = MONSTERS[index % MONSTERS.length];
      return {name:base.name, emoji:base.emoji, maxHp:base.hp + Math.floor(index / MONSTERS.length) * 20};
    }
    function makePlayer() {
      const monster = monsterFor(0);
      return {
        deck:shuffledQuestions(), index:0, selection:[], hp:MAX_HP,
        monsterIndex:0, monsterHp:monster.maxHp, score:0, combo:0,
        power:20, correct:0, wrong:0, defeated:0, busy:false, feedback:"選標點，然後按確定！"
      };
    }
    function question(p) { return p.deck[p.index]; }
    function updateQuestion(node, p) {
      node.replaceChildren();
      const q = question(p);
      if (!q) {
        node.textContent = "全部題目完成！";
        return;
      }
      const pieces = q.text.split("___");
      pieces.forEach((piece, i) => {
        node.appendChild(document.createTextNode(piece));
        if (i < pieces.length - 1) {
          const blank = document.createElement("span");
          blank.className = "blank";
          blank.textContent = p.selection[i] || "___";
          node.appendChild(blank);
        }
      });
    }
    function renderPlayer(i) {
      const p = players[i], root = views[i];
      const monster = monsterFor(p.monsterIndex);
      root.querySelector(".score").textContent = p.score;
      root.querySelector(".combo").textContent = p.combo;
      root.querySelector(".power").textContent = p.power;
      root.querySelector(".question-count").textContent = `${Math.min(p.index + 1, 30)} / ${QUESTIONS.length}`;
      root.querySelector(".monster-name").textContent = monster.name;
      root.querySelector(".monster-hp").textContent = `${p.monsterHp} / ${monster.maxHp}`;
      root.querySelector(".player-hp").textContent = `${p.hp} / ${MAX_HP}`;
      root.querySelector(".monster-fill").style.width = `${p.monsterHp / monster.maxHp * 100}%`;
      root.querySelector(".player-fill").style.width = `${p.hp / MAX_HP * 100}%`;
      root.querySelector(".meter").setAttribute("aria-valuemax", String(monster.maxHp));
      root.querySelector(".meter").setAttribute("aria-valuenow", String(p.monsterHp));
      root.querySelector(".player-meter").setAttribute("aria-valuenow", String(p.hp));
      const sprite = root.querySelector(".monster-sprite");
      sprite.firstChild.textContent = monster.emoji;
      sprite.setAttribute("aria-label", monster.name);
      updateQuestion(root.querySelector(".question-text"), p);
      const q = question(p);
      root.querySelector(".selection-hint").textContent = q
        ? `已選 ${p.selection.length} / ${q.answers.length} 個空格`
        : "等待另一位勇士完成！";
      root.querySelector(".feedback").textContent = p.feedback;
      const canPick = active && !!q && !p.busy && p.selection.length < q.answers.length;
      root.querySelectorAll(".pick").forEach(b => b.disabled = !canPick);
      root.querySelector(".cancel").disabled = !active || !q || p.busy || !p.selection.length;
      root.querySelector(".submit").disabled = !active || !q || p.busy || p.selection.length !== q.answers.length;
    }
    function render() {
      if (!players.length) return;
      players.forEach((_, i) => renderPlayer(i));
      document.getElementById("overall-progress").textContent =
        `騎士 ${players[0].index}/30 · 法師 ${players[1].index}/30`;
    }
    function startGame() {
      runId++;
      if (finishTimer) clearTimeout(finishTimer);
      finishTimer = null;
      stopSounds();
      active = true;
      players = [makePlayer(), makePlayer()];
      views.forEach(v => {
        v.classList.remove("reward");
        v.querySelectorAll(".sprite").forEach(s => s.classList.remove("hit", "attack", "defeated"));
      });
      document.getElementById("game-note").textContent = "";
      showScreen("game");
      render();
    }
    function choose(i, mark) {
      const p = players[i], q = question(p);
      if (!active || !q || p.busy || p.selection.length >= q.answers.length) return;
      p.selection.push(mark);
      p.feedback = p.selection.length === q.answers.length ? "選好了！按確定。" : "繼續選下一個標點！";
      render();
    }
    function cancelChoice(i) {
      const p = players[i];
      if (!active || p.busy || !p.selection.length) return;
      p.selection = [];
      p.feedback = "已取消，重新選吧！";
      render();
    }
    function scheduleFinish() {
      if (finishTimer || !active) return;
      finishTimer = setTimeout(() => {
        finishTimer = null;
        if (!active) return;
        if (players.some(p => p.hp === 0) ||
            players.every(p => p.index >= QUESTIONS.length && !p.busy)) {
          endGame();
        }
      }, 500);
    }
    function endGame() {
      if (!active) return;
      active = false;
      if (finishTimer) clearTimeout(finishTimer);
      finishTimer = null;
      const [a, b] = players;
      let title;
      if (a.hp === 0 && b.hp === 0) title = "兩位勇士平手！";
      else if (a.hp === 0) title = "法師勝利！";
      else if (b.hp === 0) title = "騎士勝利！";
      else if (a.score === b.score) title = "兩位勇士平手！";
      else title = a.score > b.score ? "騎士勝利！" : "法師勝利！";
      document.getElementById("result-title").textContent = title;
      [a, b].forEach((p, i) => {
        const box = document.getElementById(`p${i + 1}-results`);
        box.replaceChildren();
        [`分數：${p.score}`, `答對：${p.correct} 題`, `答錯：${p.wrong} 題`, `擊敗怪獸：${p.defeated} 隻`].forEach(line => {
          const item = document.createElement("div");
          item.textContent = line;
          item.style.marginTop = "9px";
          box.appendChild(item);
        });
      });
      showScreen("end");
      sound("victory");
    }
    async function submit(i) {
      const p = players[i], q = question(p);
      if (!active || !q || p.busy || p.selection.length !== q.answers.length) return;
      const session = runId;
      p.busy = true;
      const correct = q.answers.every((answer, n) => p.selection[n] === answer);
      const root = views[i];
      if (correct) {
        p.correct++;
        p.combo++;
        p.feedback = "答對了！準備攻擊！";
        sound("correct");
        render();
        await delay(300);
        if (session !== runId || !active) return;
        sound("attack");
        animate(root.querySelector(".hero-sprite"), "attack", 430);
        animate(root.querySelector(".monster-sprite"), "hit", 430);
        p.monsterHp = Math.max(0, p.monsterHp - p.power);
        p.score += 10;
        p.feedback = "攻擊成功！";
        if (p.combo % 3 === 0) {
          p.hp = Math.min(MAX_HP, p.hp + 20);
          p.power += 5;
          p.feedback = "連續答對 3 題！補血＋攻擊力提升！";
          animate(root, "reward", 1700);
        }
        render();
        if (p.monsterHp === 0) {
          p.defeated++;
          p.score += 20;
          p.feedback = "擊敗怪獸！下一隻來了！";
          animate(root.querySelector(".monster-sprite"), "defeated", 650);
          render();
          await delay(680);
          if (session !== runId || !active) return;
          root.querySelector(".monster-sprite").classList.remove("defeated");
          p.monsterIndex++;
          p.monsterHp = monsterFor(p.monsterIndex).maxHp;
        }
      } else {
        p.wrong++;
        p.combo = 0;
        p.hp = Math.max(0, p.hp - WRONG_DAMAGE);
        p.feedback = "再想一想！勇士扣血了。";
        sound("wrong");
        animate(root.querySelector(".hero-sprite"), "hit", 430);
        render();
        if (p.hp === 0) scheduleFinish();
        await delay(460);
        if (session !== runId || !active) return;
      }
      p.index++;
      p.selection = [];
      p.busy = false;
      if (p.index >= QUESTIONS.length && p.hp > 0) p.feedback = "全部題目完成！做得好！";
      render();
      if (players.every(player => player.index >= QUESTIONS.length)) scheduleFinish();
    }
    function goHome() {
      runId++;
      active = false;
      if (finishTimer) clearTimeout(finishTimer);
      finishTimer = null;
      stopSounds();
      showScreen("start");
    }
    function syncSoundButtons() {
      document.querySelectorAll(".sound-on").forEach(el => el.hidden = muted);
      document.querySelectorAll(".sound-off").forEach(el => el.hidden = !muted);
      document.querySelectorAll(".mute-toggle").forEach(el => el.setAttribute("aria-pressed", String(muted)));
    }
    function syncFullscreenButtons() {
      const full = !!document.fullscreenElement;
      document.querySelectorAll(".fullscreen-enter").forEach(el => el.hidden = full);
      document.querySelectorAll(".fullscreen-exit").forEach(el => el.hidden = !full);
      document.querySelectorAll(".fullscreen-toggle").forEach(el => el.setAttribute("aria-pressed", String(full)));
    }
    function showFullscreenError() {
      const note = document.getElementById(document.getElementById("game-screen").hidden ? "start-note" : "game-note");
      note.textContent = "這個畫面暫時不能全屏，仍可繼續遊戲。";
    }
    document.getElementById("start-button").addEventListener("click", startGame);
    document.getElementById("info-start").addEventListener("click", startGame);
    document.getElementById("restart-button").addEventListener("click", startGame);
    document.getElementById("instructions-button").addEventListener("click", () => showScreen("info"));
    document.getElementById("info-back").addEventListener("click", goHome);
    document.getElementById("game-home").addEventListener("click", goHome);
    document.getElementById("result-home").addEventListener("click", goHome);
    views.forEach((root, i) => {
      root.querySelectorAll(".pick").forEach(button =>
        button.addEventListener("click", () => choose(i, button.dataset.mark)));
      root.querySelector(".cancel").addEventListener("click", () => cancelChoice(i));
      root.querySelector(".submit").addEventListener("click", () => submit(i));
    });
    document.querySelectorAll(".mute-toggle").forEach(button => button.addEventListener("click", () => {
      muted = !muted;
      if (muted) stopSounds();
      syncSoundButtons();
    }));
    document.querySelectorAll(".fullscreen-toggle").forEach(button => button.addEventListener("click", async () => {
      try {
        if (document.fullscreenElement) await document.exitFullscreen();
        else if (document.documentElement.requestFullscreen) await document.documentElement.requestFullscreen();
        else showFullscreenError();
      } catch (_) { showFullscreenError(); }
      syncFullscreenButtons();
    }));
    document.addEventListener("fullscreenchange", syncFullscreenButtons);
    syncSoundButtons();
    syncFullscreenButtons();
